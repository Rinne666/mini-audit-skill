196: /**
197:  * A per-minute request limiter.
198:  *
199:  * Shape C. Whether the limit binds depends on a value read from the
200:  * environment, so the source cannot settle it.
201:  */
202: final class RateLimiter
203: {
204:     /** @var array<string, int> */
205:     private array $attempts = [];
206:
207:     public function __construct(
208:         private int $limitPerMinute,
209:     ) {
210:     }
211:
212:     public static function fromEnvironment(): self
213:     {
214:         // The effective limit depends on a deployment value.
215:         $configured = getenv('APP_RATE_LIMIT_PER_MINUTE');
216:         $limit = ($configured === false || $configured === '') ? 0 : (int) $configured;
217:         return new self($limit);
218:     }
219:
220:     public function recordAttempt(string $key): void
221:     {
222:         $this->attempts[$key] = ($this->attempts[$key] ?? 0) + 1;
223:     }
224:
225:     public function allow(string $key): bool
226:     {
227:         if ($this->limitPerMinute <= 0) {
228:             // A limit of zero means "no limit configured".
229:             return true;
230:         }
231:         return ($this->attempts[$key] ?? 0) < $this->limitPerMinute;
232:     }
233: }
