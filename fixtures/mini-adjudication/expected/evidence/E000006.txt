235: final class LegacySessionService
236: {
237:     /**
238:      * Shape D. This path used to accept a session without checking
239:      * that it was bound to the credential it was issued against. The
240:      * binding is now enforced, and the fix is visible in the source.
241:      */
242:     public function authenticate(string $sessionToken, string $currentCredential): bool
243:     {
244:         $boundCredential = $this->credentialForSession($sessionToken);
245:         if ($boundCredential === null) {
246:             return false;
247:         }
248:         // FIXED: the session is bound to the credential it was issued
249:         // against, so a rotated credential ends the session.
250:         if (!hash_equals($boundCredential, $currentCredential)) {
251:             return false;
252:         }
253:         return true;
254:     }
255:
256:     private function credentialForSession(string $sessionToken): ?string
257:     {
258:         return hash('sha256', $sessionToken);
259:     }
260: }
